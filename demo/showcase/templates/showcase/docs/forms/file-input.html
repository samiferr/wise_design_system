Restricting types and size

accept only filters the picker dialog. Enforce the real rules in clean_<field> — and validate the content type from the file, not from the filename extension. 

{% filter force_escape %}{% verbatim %}datasheet = forms.FileField(
    required=False,
    widget=forms.ClearableFileInput(attrs={"accept": ".pdf,application/pdf"}),
)

def clean_datasheet(self):
    f = self.cleaned_data.get("datasheet")
    if f and f.size > 5 * 1024 * 1024:
        raise ValidationError("Keep the file under 5 MB.")
    return f{% endverbatim %}{% endfilter %}
