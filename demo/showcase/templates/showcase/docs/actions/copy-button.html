How it works

common.js binds one delegated listener, so buttons added to the page later (in a drawer, a dialog, an HTMX swap) work without re-binding. navigator.clipboard needs a secure context; over plain HTTP the handler falls back to selecting the text so the user can copy manually. 

{% filter force_escape %}{% verbatim %}document.addEventListener('click', function (e) {
    var button = e.target.closest('.copy-button')
    if (!button) return
    var text = button.dataset.copy
    if (!text && button.dataset.copyTarget) {
        var el = document.querySelector(button.dataset.copyTarget)
        text = el ? (el.innerText || el.textContent) : ''
    }
    wiseCopy(button, text)
}){% endverbatim %}{% endfilter %}
